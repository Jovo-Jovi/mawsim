const base = process.env.BASE ?? "http://127.0.0.1:4173";

function kindOf(name) {
  if (/(^|_)draw$/.test(name)) return "draw";
  if (/(^|_)settle$/.test(name)) return "settle";
  if (/(^|_)rise$/.test(name)) return "rise";
  return null;
}

function msOf(token) {
  const value = String(token).trim();
  if (value.endsWith("ms")) return Number(value.slice(0, -2));
  if (value.endsWith("s")) return Number(value.slice(0, -1)) * 1000;
  return Number(value);
}

const browser = await chromium.launch({ headless: true });

async function readPage(path, reducedMotion) {
  const context = await browser.newContext({
    viewport: { width: 1280, height: 900 },
    reducedMotion,
  });
  const page = await context.newPage();
  await page.goto(base + path, { waitUntil: "commit" });
  const sampled = await page.evaluate(async () => {
    const seen = new Map();
    const samples = [];
    const origin = performance.now();
    const marks = [0, 400, 900, 1400];
    let mark = 0;
    while (performance.now() - origin < 1500) {
      const now = performance.now() - origin;
      const anims = document.getAnimations().filter((anim) => anim instanceof CSSAnimation);
      if (mark < marks.length && now >= marks[mark]) {
        samples.push({ atMs: Math.round(now), cssAnimations: anims.length });
        mark += 1;
      }
      for (const anim of anims) {
        const timing = anim.effect?.getComputedTiming?.() ?? {};
        const target = anim.effect?.target;
        const parts = [];
        let node = target instanceof Element ? target : null;
        while (node && node !== document.body) {
          const parent = node.parentElement;
          const index = parent ? [...parent.children].indexOf(node) : 0;
          parts.push(node.tagName.toLowerCase() + index);
          node = parent;
        }
        const key = anim.animationName + "@" + parts.join("/");
        seen.set(key, {
          animationName: anim.animationName,
          iterations: timing.iterations,
          delay: timing.delay,
          duration: timing.duration,
          playState: anim.playState,
        });
      }
      await new Promise((resolve) => requestAnimationFrame(resolve));
    }
    return { samples, seen: [...seen.values()] };
  });
  const remain = 3000 - (await page.evaluate(() => performance.now()));
  if (remain > 0) await page.waitForTimeout(remain);
  const after = await page.evaluate(() => {
    const running = document.getAnimations().filter((anim) => anim instanceof CSSAnimation && anim.playState === "running");
    const arc = document.querySelector("path.currentArc, path[class*='currentArc']")

    const style = arc ? getComputedStyle(arc) : null;

      return {

        animationDuration: computed.animationDuration,
        animationDelay: computed.animationDelay,
        animationIterationCount: computed.animationIterationCount,

            strokeDashoffset: style.strokeDashoffset,
            opacity: style.opacity,

      reduced: matchMedia("(prefers-reduced-motion: reduce)").matches,

function summarise(result) {
  const groups = { draw: [], settle: [], rise: [], other: [] };
  for (const row of result.seen) {
    const kind = kindOf(row.animationName) ?? "other";
    groups[kind].push(row);
  }
  const span = (rows) => rows.map((row) => Number(row.delay) + Number(row.duration));
  const declaredKinds = { draw: 0, settle: 0, rise: 0 };
  for (const row of result.after.declared) {
    const names = row.animationName.split(",").map((name) => name.trim());
    const durations = row.animationDuration.split(",").map((token) => msOf(token));
    const delays = row.animationDelay.split(",").map((token) => msOf(token));
    const iterations = row.animationIterationCount.split(",").map((token) => token.trim());
    const iterationFor = (index) => iterations[index] ?? iterations[0];
    names.forEach((name, index) => {
      const kind = kindOf(name);
      if (!kind) return;
      declaredKinds[kind] += 1;
      groups[kind + "Declared"] ??= [];
      groups[kind + "Declared"].push({
        delay: delays[index] ?? delays[0],
        duration: durations[index] ?? durations[0],
        iterations: iterationFor(index),
        sum: (delays[index] ?? delays[0]) + (durations[index] ?? durations[0]),
      });
    });
  }
  const drawSpans = span(groups.draw);
  const declaredDrawSpans = (groups.drawDeclared ?? []).map((row) => row.sum);
  return {
    path: result.path,
    reducedMotion: result.reducedMotion,
    mediaReduced: result.after.reduced,
    samples: result.samples,
    observed: {
      draw: groups.draw.length,
      settle: groups.settle.length,
      rise: groups.rise.length,
      other: groups.other.map((row) => row.animationName),
      iterations: [...new Set([...groups.draw, ...groups.settle, ...groups.rise].map((row) => row.iterations))],
      names: [...new Set([...groups.draw, ...groups.settle, ...groups.rise].map((row) => row.animationName))],
      largestDrawDelayPlusDuration: drawSpans.length ? Math.max(...drawSpans) : null,
