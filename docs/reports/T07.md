import { createRequire } from "node:module";
import { pathToFileURL } from "node:url";

const require = createRequire(pathToFileURL(`${process.cwd()}/package.json`));
const { chromium } = require("@playwright/test");

function lin(c) {
  const s = c / 255;
  return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
}
function lum(rgb) {
  return 0.2126 * lin(rgb[0]) + 0.7152 * lin(rgb[1]) + 0.0722 * lin(rgb[2]);
}
function ratio(a, b) {
  const l1 = lum(a);
  const l2 = lum(b);
  const light = Math.max(l1, l2);

function parse(color) {
  if (!color) return null;
  const text = color.trim();
  const hex = text.match(/^#([0-9a-f]{3,8})$/i);
  if (hex) {
    let h = hex[1];
    if (h.length === 3 || h.length === 4) h = [...h].map((c) => c + c).join("");
    const n = parseInt(h.slice(0, 6), 16);
    const rgb = [(n >> 16) & 255, (n >> 8) & 255, n & 255];
    const alpha = h.length === 8 ? parseInt(h.slice(6, 8), 16) / 255 : 1;
    return { rgb, alpha };
  }
  const rgb = text.match(/rgba?\(([^)]+)\)/i);
  if (rgb) {
    const p = rgb[1].split(/[,\s/]+/).filter(Boolean).map(Number);
    return { rgb: p.slice(0, 3), alpha: p.length > 3 ? p[3] : 1 };
  }
  const srgb = text.match(/color\(srgb\s+([0-9.]+)\s+([0-9.]+)\s+([0-9.]+)/i);
  if (srgb) return { rgb: [1, 2, 3].map((i) => Math.round(Number(srgb[i]) * 255)), alpha: 1 };
  return null;
}

  return fg.map((c, i) => Math.round(c * a + bg[i] * (1 - a)));
}
function mix(a, aw, b) {
  return a.map((c, i) => Math.round(c * aw + b[i] * (1 - aw)));
}

function lum(rgb) {

  const dark = Math.min(l1, l2);

function parse(color) {
  if (!color) return null;
  const text = color.trim();
  const hex = text.match(/^#([0-9a-f]{3,8})$/i);
  if (hex) {
    let h = hex[1];
    if (h.length === 3 || h.length === 4) h = [...h].map((c) => c + c).join("");
    const n = parseInt(h.slice(0, 6), 16);
